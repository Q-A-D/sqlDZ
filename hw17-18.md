```
USE ShopDB;
GO

INSERT INTO Customers (CustomerID, FirstName, LastName)
SELECT TOP 100 BusinessEntityID, FirstName, LastName 
FROM AdventureWorks2022.Person.Person;
GO

INSERT INTO Categories (CategoryID, CategoryName)
SELECT TOP 100 ProductCategoryID, Name 
FROM AdventureWorks2022.Production.ProductCategory;
GO

INSERT INTO Products (ProductID, ProductName, Price, CategoryID)
SELECT TOP 100 ProductID, Name, ListPrice, ProductSubcategoryID 
FROM AdventureWorks2022.Production.Product;
GO

INSERT INTO Employees (EmployeeID, FirstName, LastName, Position)
SELECT TOP 100 p.BusinessEntityID, p.FirstName, p.LastName, e.JobTitle
FROM AdventureWorks2022.Person.Person p
JOIN AdventureWorks2022.HumanResources.Employee e ON p.BusinessEntityID = e.BusinessEntityID;
GO

INSERT INTO Orders (OrderID, CustomerID, EmployeeID, OrderDate, TotalAmount)
SELECT TOP 100 SalesOrderID, CustomerID, SalesPersonID, OrderDate, TotalDue
FROM AdventureWorks2022.Sales.SalesOrderHeader;
GO

INSERT INTO OrderDetails (OrderDetailID, OrderID, ProductID, Quantity, UnitPrice)
SELECT TOP 100 SalesOrderDetailID, SalesOrderID, ProductID, OrderQty, UnitPrice
FROM AdventureWorks2022.Sales.SalesOrderDetail;
GO
```
